  function classifyRisk(t){
    const mins=t.cptMs?(t.cptMs-Date.now())/60000:null;t.minutesToCpt=mins;const reasons=[];
    if(t.ownerQty.unowned){reasons.push(`UNOWNED ${t.ownerQty.unowned}`);}if(t.ownerQty.whse){reasons.push(`WHSE ${t.ownerQty.whse}`);}if(t.ownerQty.workOrder){reasons.push(`WORK ORDER ${t.ownerQty.workOrder}`);}if(t.flags.includes('CS_MULTI_CHUTE'))reasons.push('SAME CS MULTI-CHUTE');if(t.flags.includes('MULTI_CS'))reasons.push('MULTI CS');
    if(reasons.length){t.riskReason=reasons.join(' · ');return['ACT NOW',100];}
    if(t.truth==='RODEO_BATCH_ERROR'||t.flags.includes('FC_READ_ERROR')){t.riskReason=t.error||'Read failure';return['VERIFY',78];}
    if(mins!=null&&mins<0){t.riskReason=`CPT overdue ${Math.abs(Math.round(mins))}m`;return['ACT NOW',100];}
    if(t.condition==='C15'&&mins!=null&&mins<=60){t.riskReason=`C15 · ${Math.round(mins)}m to CPT`;return['ACT NOW',96];}
    if(mins!=null&&mins<=30){t.riskReason=`${Math.round(mins)}m to CPT`;return['ACT NOW',92];}
    if(mins!=null&&mins<=60){t.riskReason=`${Math.round(mins)}m to CPT`;return['WATCH',72];}
    if(mins!=null&&mins<=90){t.riskReason=`${Math.round(mins)}m to CPT`;return['WATCH',55];}
    if(t.condition==='C5'){t.riskReason='C5 cleanup';return['CLEANUP',30];}
    if(t.occupied){t.riskReason=t.condition||'Occupied';return['WATCH',35];}
    if(t.truth==='FC_CONFIRMED_EMPTY'){t.riskReason='FC confirmed empty';return['CLEAR',0];}
    t.riskReason='Not FC verified';return['VERIFY',20];
  }
