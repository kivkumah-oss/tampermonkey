    const vals=[...state.chutes.values()];const known=vals.filter(t=>['RODEO_OCCUPIED','FC_CONFIRMED_EMPTY','FC_CONFIRMED_OCCUPIED'].includes(t.truth)).length;const occupied=vals.filter(t=>t.occupied).length;const empty=vals.filter(t=>t.truth==='FC_CONFIRMED_EMPTY').length;const act=vals.filter(t=>t.risk==='ACT NOW').length;const watch90=vals.filter(t=>t.minutesToCpt!=null&&t.minutesToCpt<=90&&t.minutesToCpt>=0).length;const watch60=vals.filter(t=>t.minutesToCpt!=null&&t.minutesToCpt<=60&&t.minutesToCpt>=0).length;const watch30=vals.filter(t=>t.minutesToCpt!=null&&t.minutesToCpt<=30&&t.minutesToCpt>=0).length;const c4=vals.filter(t=>t.condition==='C4').length;const c15=vals.filter(t=>t.condition==='C15').length;const c5=vals.filter(t=>t.condition==='C5').length;const anomalies=vals.filter(t=>t.ownerQty.unowned||t.ownerQty.whse||t.ownerQty.workOrder||t.flags.includes('CS_MULTI_CHUTE')||t.flags.includes('MULTI_CS')||t.error).length;return{total:vals.length,known,coverage:Math.round(known/vals.length*100),occupied,empty,act,watch90,watch60,watch30,c4,c15,c5,anomalies};
  }

  function pathSummary(){const m=new Map();for(const t of state.chutes.values()){if(!t.occupied)continue;const p=t.processPath||'UNKNOWN';const x=m.get(p)||{path:p,chutes:0,units:0,act:0};x.chutes++;x.units+=t.totalQty||0;if(t.risk==='ACT NOW')x.act++;m.set(p,x);}return[...m.values()].sort((a,b)=>b.act-a.act||b.units-a.units||a.path.localeCompare(b.path));}
  function queueItems(){return[...state.chutes.values()].filter(t=>t.risk!=='CLEAR'&&focusMatch(t)).sort((a,b)=>b.riskRank-a.riskRank||(a.cptMs||Infinity)-(b.cptMs||Infinity)).slice(0,16);}
  function sourceBadge(t){return t.truth==='RODEO_OCCUPIED'?'R':t.truth.startsWith('FC_')?'F':t.truth==='RODEO_BATCH_ERROR'?'!':'?';}
  function tileClass(t){if(t.error)return'read-error';if(t.ownerQty.unowned||t.ownerQty.whse||t.ownerQty.workOrder||t.flags.includes('CS_MULTI_CHUTE')||t.flags.includes('MULTI_CS'))return'anomaly';if(t.condition==='C4')return'c4';if(t.condition==='C5')return'c5';if(t.condition==='C15')return'c15';if(t.occupied)return'occupied';if(t.truth==='FC_CONFIRMED_EMPTY')return'free';return'unverified';}
  function urgencyClass(t){if(t.minutesToCpt==null)return'';if(t.minutesToCpt<=30)return'urgent30';if(t.minutesToCpt<=60)return'urgent60';if(t.minutesToCpt<=90)return'urgent90';return'';}
  function focusMatch(t){if(state.focus==='overview')return true;if(state.focus==='action')return['ACT NOW','WATCH','CLEANUP','VERIFY'].includes(t.risk);if(state.focus==='urgent')return t.minutesToCpt!=null&&t.minutesToCpt<=90;if(state.focus==='c4')return t.condition==='C4';if(state.focus==='c15')return t.condition==='C15';if(state.focus==='c5')return t.condition==='C5';if(state.focus==='anomalies')return t.error||t.flags.length>0||t.ownerQty.unowned||t.ownerQty.whse||t.ownerQty.workOrder;return true;}
  function why(t){const bits=[];if(t.minutesToCpt!=null)bits.push(`${Math.round(t.minutesToCpt)}m`);if(t.condition)bits.push(t.condition);if(t.processPath)bits.push(t.processPath);if(t.riskReason)bits.push(t.riskReason);if(t.runner)bits.push(t.runner);return[...new Set(bits)].join(' · ')||t.truth;}

  function card(tone,title,value,note,focus){return`<button class="card ${tone} ${state.focus===focus?'active':''}" data-focus="${focus}"><div class="eyebrow">${esc(title)}</div><div class="value">${esc(value)}</div><div class="note">${esc(note)}</div></button>`;}

  function bayHtml(bay,slots){
    const expanded=state.expandedBays.has(bay);const b=[...state.chutes.values()].filter(t=>t.bay===bay);const occ=b.filter(t=>t.occupied).length;const act=b.filter(t=>t.risk==='ACT NOW').length;const unknown=b.filter(t=>!['RODEO_OCCUPIED','FC_CONFIRMED_EMPTY','FC_CONFIRMED_OCCUPIED'].includes(t.truth)).length;
    let rows='';if(expanded){for(const row of WALL.rows){rows+=`<div class="bayRow"><div class="bayRowLabel">${row}</div><div class="bayRowChutes">`;for(const slot of slots){const id=`${WALL.prefix}${bay}${row}${slot}`;const t=state.chutes.get(id)||blankTruth({id,row,bay,slot});rows+=`<button class="chute ${tileClass(t)} ${urgencyClass(t)} ${state.selected===id?'selected':''} ${state.focus!=='overview'&&!focusMatch(t)?'dimmed':''}" data-chute="${id}" title="${esc(why(t))}"><span class="src">${sourceBadge(t)}</span><span class="slot">${row}${slot}</span><small>${t.error?'ERR':t.totalQty||0}</small></button>`;}rows+='</div></div>';}}
    return`<div class="bay"><div class="bayHead" data-bay="${bay}"><div><div class="bayName">Bay ${bay}</div><div class="baySub">${occ}/${b.length} occupied · ${unknown} unverified${act?` · ${act} ACT NOW`:''}</div></div><span class="chev">${expanded?'−':'+'}</span></div>${expanded?`<div class="bayGridWrap">${rows}</div>`:''}</div>`;
  }

  function wallHtml(){return WALL.bays.map(([bay,slots])=>bayHtml(bay,slots)).join('');}
